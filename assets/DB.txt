  username      TEXT NOT NULL UNIQUE,
  password_hash TEXT NOT NULL,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- 開啟 RLS：預設拒絕所有 anon/public 角色存取
ALTER TABLE public.users ENABLE ROW LEVEL SECURITY;

-- 刻意不建立任何 SELECT / INSERT policy：
-- 前端 publishable key 無法直接讀寫本表；所有存取都由後端用 service-role 金鑰繞過 RLS 完成。

-- 補 service_role 對 users 表的存取權限（後端註冊／登入使用）
GRANT SELECT, INSERT ON public.users TO service_role;
GRANT USAGE, SELECT ON SEQUENCE public.users_id_seq TO service_role;

-- 1. 補存檔與排行欄位（users 表已存在；既有列自動帶預設值）
ALTER TABLE public.users
  ADD COLUMN IF NOT EXISTS levels_completed  INTEGER      NOT NULL DEFAULT 0,
  ADD COLUMN IF NOT EXISTS score_per_level   JSONB        NOT NULL DEFAULT '[]'::jsonb,
  ADD COLUMN IF NOT EXISTS stars_per_level   JSONB        NOT NULL DEFAULT '[]'::jsonb,
  ADD COLUMN IF NOT EXISTS total_score       INTEGER      NOT NULL DEFAULT 0,
  ADD COLUMN IF NOT EXISTS updated_at        TIMESTAMPTZ  NOT NULL DEFAULT NOW();

-- 2. 排行榜索引：總積分由高到低，同分時先註冊（id 小）在前
CREATE INDEX IF NOT EXISTS users_total_score_idx
  ON public.users (total_score DESC, id ASC);

-- 3. 寫入／更新時自動重算 total_score 並刷新 updated_at
CREATE OR REPLACE FUNCTION public.sync_users_save()
RETURNS TRIGGER AS $$
BEGIN
  NEW.total_score := COALESCE((
    SELECT SUM(elem::INTEGER)::INTEGER
    FROM jsonb_array_elements(COALESCE(NEW.score_per_level, '[]'::jsonb)) AS elem
  ), 0);
  NEW.updated_at := NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS users_sync_save ON public.users;
CREATE TRIGGER users_sync_save
  BEFORE INSERT OR UPDATE ON public.users
  FOR EACH ROW EXECUTE FUNCTION public.sync_users_save();

-- 4. RLS：維持開啟，刻意不建立任何 anon/public policy
ALTER TABLE public.users ENABLE ROW LEVEL SECURITY;

-- 1. 資料表（全新建立；已存在則保留原表，不覆寫）
CREATE TABLE IF NOT EXISTS public.users (
  id               BIGINT GENERATED ALWAYS AS IDENTITY,
  username         TEXT NOT NULL,
  password_hash    TEXT NOT NULL,
  created_at       TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  levels_completed INTEGER      NOT NULL DEFAULT 0,
  score_per_level  JSONB        NOT NULL DEFAULT '[]'::jsonb,
  stars_per_level  JSONB        NOT NULL DEFAULT '[]'::jsonb,
  total_score      INTEGER      NOT NULL DEFAULT 0,
  updated_at       TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
  CONSTRAINT users_pkey          PRIMARY KEY (id),
  CONSTRAINT users_username_key  UNIQUE (username)
);

-- 2. 舊表升級：補缺少的存檔欄位（可重複執行）
ALTER TABLE public.users
  ADD COLUMN IF NOT EXISTS levels_completed  INTEGER      NOT NULL DEFAULT 0,
  ADD COLUMN IF NOT EXISTS score_per_level   JSONB        NOT NULL DEFAULT '[]'::jsonb,
  ADD COLUMN IF NOT EXISTS stars_per_level   JSONB        NOT NULL DEFAULT '[]'::jsonb,
  ADD COLUMN IF NOT EXISTS total_score       INTEGER      NOT NULL DEFAULT 0,
  ADD COLUMN IF NOT EXISTS updated_at        TIMESTAMPTZ   NOT NULL DEFAULT NOW();

-- 3. 舊表若缺 username 唯一約束，補上（已存在則略過）
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint
    WHERE conname = 'users_username_key' AND conrelid = 'public.users'::regclass
  ) THEN
    ALTER TABLE public.users ADD CONSTRAINT users_username_key UNIQUE (username);
  END IF;
END $$;

-- 4. 排行榜索引：總積分由高到低，同分時先註冊（id 小）在前
CREATE INDEX IF NOT EXISTS users_total_score_idx
  ON public.users (total_score DESC, id ASC);

-- 5. 觸發器函數：寫入／更新時自動重算 total_score 並刷新 updated_at
CREATE OR REPLACE FUNCTION public.sync_users_save()
RETURNS TRIGGER AS $$
BEGIN
  NEW.total_score := COALESCE((
    SELECT SUM(GREATEST(0, (elem #>> '{}')::numeric))::INTEGER
    FROM jsonb_array_elements(COALESCE(NEW.score_per_level, '[]'::jsonb)) AS elem
  ), 0);
  NEW.updated_at := NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql VOLATILE;

-- 6. 綁定觸發器（先卸舊再建，可重複執行）
DROP TRIGGER IF EXISTS users_sync_save ON public.users;
CREATE TRIGGER users_sync_save
  BEFORE INSERT OR UPDATE ON public.users
  FOR EACH ROW
  EXECUTE FUNCTION public.sync_users_save();

-- 7. RLS：維持開啟，刻意不建立任何 anon/public policy
ALTER TABLE public.users ENABLE ROW LEVEL SECURITY;

-- 8. service_role（後端 secret 金鑰）資料表與識別序列權限
GRANT SELECT, INSERT, UPDATE, DELETE ON public.users TO service_role;
GRANT USAGE, SELECT ON SEQUENCE public.users_id_seq TO service_role;
